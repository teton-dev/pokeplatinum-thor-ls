# Gen 5 battles, widescreen: the arena's background and the sprites' placement while the camera idles. A wild
# battle's command menu, shot ten times a second apart. From the turtwig fixture:
#
#   TEST_CONFIG='SecondaryScreen=true Widescreen=true WindowWidth=1280 WindowHeight=720' \
#       tools/docker/desktop.sh test tests/scripts/gen5_wide_bg.txt --from turtwig

wait 30

until battle
until command
wait 60

wait 45

wait 45
shot bg-3
wait 45
shot bg-4
wait 45
shot bg-5
